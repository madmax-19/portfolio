PROJECT 04 / EMBEDDED SYSTEMS

MIND
GAME.

An Arduino-based memory challenge that combines four LEDs, four push buttons, a speaker and a 128×64 SH1106 OLED display. The game generates a random sequence, shows it through light and sound, and increases the level after every successful round.

01 / OVERVIEW

A memory game built
with simple hardware.

The idea

The player presses START and watches a sequence of four possible LED and sound signals. The player then reproduces the same sequence using the four buttons. Every successful round adds one more signal to the sequence, making the game progressively harder.

Core features

  • Random sequence generation up to 100 levels.
  • Four LEDs provide visual feedback.
  • Four tones identify the four game inputs.
  • SH1106 OLED displays game status and score.
  • Wrong input ends the round and shows GAME OVER.

04 / GAME LOGIC

Each round adds
one more challenge.

Sequence generation

At the beginning of a new game, the program fills the sequence array with random values from 0 to 3. The current level decides how many values are shown during the round.

LEVEL 1 → 1 signal
LEVEL 2 → 2 signals
LEVEL 3 → 3 signals
LEVEL 4 → 4 signals

Player validation

Each button maps to one sequence value. The player input is checked immediately. A correct sequence increases the score and level. A wrong button returns false and triggers the game-over screen.

CORRECT → score++ → level++
WRONG → GAME OVER
START → reset score + level

06 / COMPLETE ARDUINO CODE

The actual Mind Game code.

#include <U8g2lib.h>

// ================= OLED =================

U8G2_SH1106_128X64_NONAME_F_4W_HW_SPI u8g2(
  U8G2_R0,
  /* CS   */ 10,
  /* DC   */ 9,
  /* RESET */ 8
);

// ================= PINS =================

// Buttons
const int buttonPins[4] = {2, 3, 4, 5};

// Start button
const int startButton = 6;

// LEDs
const int ledPins[4] = {
  A0, A1, A2, A3
};

// Speaker / buzzer
const int speakerPin = 12;

// ================= GAME SETTINGS =================

const int MAX_SEQUENCE = 100;

int sequence[MAX_SEQUENCE];

int level = 1;
int score = 0;

// Different tones for the four buttons
int tones[4] = {
  262,
  330,
  392,
  523
};

// =================================================
// OLED MESSAGE
// =================================================

void showMessage(String line1, String line2 = "", String line3 = "") {

  u8g2.clearBuffer();

  u8g2.setFont(u8g2_font_ncenB08_tr);

  u8g2.drawStr(0, 18, line1.c_str());

  if (line2 != "") {
    u8g2.drawStr(0, 38, line2.c_str());
  }

  if (line3 != "") {
    u8g2.drawStr(0, 58, line3.c_str());
  }

  u8g2.sendBuffer();
}

// =================================================
// LED + SOUND
// =================================================

void playSignal(int index) {

  digitalWrite(ledPins[index], HIGH);

  tone(speakerPin, tones[index]);

  delay(400);

  digitalWrite(ledPins[index], LOW);

  noTone(speakerPin);

  delay(150);
}

// =================================================
// SHOW RANDOM SEQUENCE
// =================================================

void showSequence() {

  showMessage(
    "MEMORIZE!",
    "Watch the",
    "sequence..."
  );

  delay(800);

  for (int i = 0; i < level; i++) {

    playSignal(sequence[i]);

    delay(100);
  }
}

// =================================================
// WAIT FOR PLAYER BUTTON
// =================================================

int waitForButton() {

  while (true) {

    for (int i = 0; i < 4; i++) {

      if (digitalRead(buttonPins[i]) == LOW) {

        playSignal(i);

        // Wait until button is released
        while (digitalRead(buttonPins[i]) == LOW) {
          delay(10);
        }

        return i;
      }
    }

    delay(5);
  }
}

// =================================================
// PLAYER TURN
// =================================================

bool playerTurn() {

  showMessage(
    "YOUR TURN",
    ("LEVEL: " + String(level)).c_str(),
    "Repeat sequence"
  );

  delay(500);

  for (int i = 0; i < level; i++) {

    int input = waitForButton();

    // Wrong button
    if (input != sequence[i]) {

      return false;
    }
  }

  return true;
}

// =================================================
// CORRECT ANSWER
// =================================================

void successScreen() {

  showMessage(
    "CORRECT!",
    ("LEVEL " + String(level)).c_str(),
    "Get ready..."
  );

  tone(speakerPin, 784);
  delay(150);

  tone(speakerPin, 988);
  delay(150);

  noTone(speakerPin);

  delay(800);
}

// =================================================
// GAME OVER
// =================================================

void gameOver() {

  tone(speakerPin, 180);

  delay(500);

  noTone(speakerPin);

  showMessage(
    "GAME OVER",
    ("SCORE: " + String(score)).c_str(),
    ("LEVEL: " + String(level)).c_str()
  );

  delay(2500);

  showMessage(
    "PRESS START",
    "TO PLAY AGAIN"
  );
}

// =================================================
// WAIT FOR START
// =================================================

void waitForStart() {

  showMessage(
    "MIND GAME",
    "Memory Challenge",
    "PRESS START"
  );

  while (digitalRead(startButton) == HIGH) {
    delay(20);
  }

  while (digitalRead(startButton) == LOW) {
    delay(20);
  }

  delay(300);
}

// =================================================
// START NEW GAME
// =================================================

void newGame() {

  level = 1;
  score = 0;

  // Generate random sequence
  for (int i = 0; i < MAX_SEQUENCE; i++) {

    sequence[i] = random(0, 4);
  }

  showMessage(
    "GET READY!",
    "Game starting..."
  );

  // Start melody
  tone(speakerPin, 523);
  delay(150);

  tone(speakerPin, 659);
  delay(150);

  tone(speakerPin, 784);
  delay(150);

  noTone(speakerPin);

  delay(800);
}

// =================================================
// SETUP
// =================================================

void setup() {

  // Start OLED
  u8g2.begin();

  // Buttons and LEDs
  for (int i = 0; i < 4; i++) {

    pinMode(buttonPins[i], INPUT_PULLUP);

    pinMode(ledPins[i], OUTPUT);

    digitalWrite(ledPins[i], LOW);
  }

  // Start button
  pinMode(startButton, INPUT_PULLUP);

  // Speaker
  pinMode(speakerPin, OUTPUT);

  // Random seed
  randomSeed(analogRead(A5));

  // Startup screen
  showMessage(
    "MIND GAME",
    "Arduino Memory",
    "Initializing..."
  );

  delay(1500);
}

// =================================================
// MAIN LOOP
// =================================================

void loop() {

  // Wait for START
  waitForStart();

  // Reset game
  newGame();

  // Game loop
  while (level <= MAX_SEQUENCE) {

    // Show sequence
    showSequence();

    // Player repeats sequence
    if (playerTurn()) {

      // Correct
      score++;

      successScreen();

      level++;

    } else {

      // Wrong
      gameOver();

      break;
    }
  }
}
