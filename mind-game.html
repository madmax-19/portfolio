01 / PROJECT OVERVIEW

A physical memory game built with Arduino.

Mind Game is an Arduino-based memory challenge. The system generates a random sequence and presents it to the player using four LEDs and four different sound tones.

The player must remember the sequence and reproduce it using four physical buttons. When the sequence is completed correctly, the score and level increase and the next round becomes longer.

03 / HOW IT WORKS

Watch it. Remember it. Repeat it.

Game concept

After the player presses START, the Arduino generates a random sequence.

Each value in the sequence corresponds to one of four LEDs and one of four different audio tones.

The player watches and listens to the sequence, then reproduces it by pressing the matching buttons.

Main features

Random sequence generation
Progressive difficulty
Four LED indicators
Four different tones
OLED interface
Score and level tracking

07 / GAME LOGIC

Every correct answer makes the next round harder.

Progressive sequence

The game starts at level 1.

After successfully reproducing the complete sequence, the score increases and the level is incremented.

The next round therefore contains one additional sequence value.

Level 1 → 1 signal
Level 2 → 2 signals
Level 3 → 3 signals
Level 4 → 4 signals

Game over

During the player's turn, every button press is compared with the expected sequence value.

If the player presses the wrong button, `playerTurn()` returns false.

The game then calls `gameOver()`, plays the game-over tone and displays the score and current level on the OLED.

08 / COMPLETE ARDUINO CODE

The actual Mind Game code.

mind_game.ino 
#include <U8g2lib.h>

// ================= OLED =================

U8G2_SH1106_128X64_NONAME_F_4W_HW_SPI u8g2(
  U8G2_R0,
  /* CS   */ 10,
  /* DC   */ 9,
  /* RESET */ 8
);

// ================= PINS =================

// Buttons
const int buttonPins[4] = {2, 3, 4, 5};

// Start button
const int startButton = 6;

// LEDs
const int ledPins[4] = {
  A0, A1, A2, A3
};

// Speaker / buzzer
const int speakerPin = 12;

// ================= GAME SETTINGS =================

const int MAX_SEQUENCE = 100;

int sequence[MAX_SEQUENCE];

int level = 1;
int score = 0;

// Different tones for the four buttons
int tones[4] = {
  262,
  330,
  392,
  523
};


// =================================================
// OLED MESSAGE
// =================================================

void showMessage(String line1, String line2 = "", String line3 = "") {

  u8g2.clearBuffer();

  u8g2.setFont(u8g2_font_ncenB08_tr);

  u8g2.drawStr(0, 18, line1.c_str());

  if (line2 != "") {
    u8g2.drawStr(0, 38, line2.c_str());
  }

  if (line3 != "") {
    u8g2.drawStr(0, 58, line3.c_str());
  }

  u8g2.sendBuffer();
}


// =================================================
// LED + SOUND
// =================================================

void playSignal(int index) {

  digitalWrite(ledPins[index], HIGH);

  tone(speakerPin, tones[index]);

  delay(400);

  digitalWrite(ledPins[index], LOW);

  noTone(speakerPin);

  delay(150);
}


// =================================================
// SHOW RANDOM SEQUENCE
// =================================================

void showSequence() {

  showMessage(
    "MEMORIZE!",
    "Watch the",
    "sequence..."
  );

  delay(800);

  for (int i = 0; i < level; i++) {

    playSignal(sequence[i]);

    delay(100);
  }
}


// =================================================
// WAIT FOR PLAYER BUTTON
// =================================================

int waitForButton() {

  while (true) {

    for (int i = 0; i < 4; i++) {

      if (digitalRead(buttonPins[i]) == LOW) {

        playSignal(i);

        // Wait until button is released
        while (digitalRead(buttonPins[i]) == LOW) {
          delay(10);
        }

        return i;
      }
    }

    delay(5);
  }
}


// =================================================
// PLAYER TURN
// =================================================

bool playerTurn() {

  showMessage(
    "YOUR TURN",
    ("LEVEL: " + String(level)).c_str(),
    "Repeat sequence"
  );

  delay(500);

  for (int i = 0; i < level; i++) {

    int input = waitForButton();

    // Wrong button
    if (input != sequence[i]) {

      return false;
    }
  }

  return true;
}


// =================================================
// CORRECT ANSWER
// =================================================

void successScreen() {

  showMessage(
    "CORRECT!",
    ("LEVEL " + String(level)).c_str(),
    "Get ready..."
  );

  tone(speakerPin, 784);
  delay(150);

  tone(speakerPin, 988);
  delay(150);

  noTone(speakerPin);

  delay(800);
}


// =================================================
// GAME OVER
// =================================================

void gameOver() {

  tone(speakerPin, 180);

  delay(500);

  noTone(speakerPin);

  showMessage(
    "GAME OVER",
    ("SCORE: " + String(score)).c_str(),
    ("LEVEL: " + String(level)).c_str()
  );

  delay(2500);

  showMessage(
    "PRESS START",
    "TO PLAY AGAIN"
  );
}


// =================================================
// WAIT FOR START
// =================================================

void waitForStart() {

  showMessage(
    "MIND GAME",
    "Memory Challenge",
    "PRESS START"
  );

  while (digitalRead(startButton) == HIGH) {
    delay(20);
  }

  while (digitalRead(startButton) == LOW) {
    delay(20);
  }

  delay(300);
}


// =================================================
// START NEW GAME
// =================================================

void newGame() {

  level = 1;
  score = 0;

  // Generate random sequence
  for (int i = 0; i < MAX_SEQUENCE; i++) {

    sequence[i] = random(0, 4);
  }

  showMessage(
    "GET READY!",
    "Game starting..."
  );

  // Start melody
  tone(speakerPin, 523);
  delay(150);

  tone(speakerPin, 659);
  delay(150);

  tone(speakerPin, 784);
  delay(150);

  noTone(speakerPin);

  delay(800);
}


// =================================================
// SETUP
// =================================================

void setup() {

  // Start OLED
  u8g2.begin();

  // Buttons and LEDs
  for (int i = 0; i < 4; i++) {

    pinMode(buttonPins[i], INPUT_PULLUP);

    pinMode(ledPins[i], OUTPUT);

    digitalWrite(ledPins[i], LOW);
  }

  // Start button
  pinMode(startButton, INPUT_PULLUP);

  // Speaker
  pinMode(speakerPin, OUTPUT);

  // Random seed
  randomSeed(analogRead(A5));

  // Startup screen
  showMessage(
    "MIND GAME",
    "Arduino Memory",
    "Initializing..."
  );

  delay(1500);
}


// =================================================
// MAIN LOOP
// =================================================

void loop() {

  // Wait for START
  waitForStart();

  // Reset game
  newGame();

  // Game loop
  while (level <= MAX_SEQUENCE) {

    // Show sequence
    showSequence();

    // Player repeats sequence
    if (playerTurn()) {

      // Correct
      score++;

      successScreen();

      level++;

    } else {

      // Wrong
      gameOver();

      break;
    }
  }
}
